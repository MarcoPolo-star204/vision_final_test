cmake_minimum_required(VERSION 3.0.2)
project(ft_selector)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp ft_msgs geometry_msgs)
catkin_package(CATKIN_DEPENDS roscpp ft_msgs geometry_msgs)
include_directories(${catkin_INCLUDE_DIRS})
add_executable(ft_selector_node src/ft_selector_node.cpp)
add_dependencies(ft_selector_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(ft_selector_node ${catkin_LIBRARIES})

cmake_minimum_required(VERSION 3.0.2)
project(ft_tracker)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp ft_msgs geometry_msgs visualization_msgs)
find_package(OpenCV REQUIRED)
catkin_package(CATKIN_DEPENDS roscpp ft_msgs geometry_msgs visualization_msgs)
include_directories(${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
add_executable(ft_tracker_node src/ft_tracker_node.cpp)
add_dependencies(ft_tracker_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(ft_tracker_node ${catkin_LIBRARIES} ${OpenCV_LIBS})
